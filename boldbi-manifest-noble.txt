ARG TARGETARCH
ARG TARGETPLATFORM
ARG INSTALL_POSTGRES=false

# ====================================================
# Common Packages
# (package names adjusted for Ubuntu 24.04 "noble":
#  libasound2 -> libasound2t64, libgcc1 -> libgcc-s1,

# ====================================================

    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# ====================================================
# ARM64 Chromium
# (noble's chromium-browser apt package is a snap stub that
#  fails inside containers; install the real arm64 build
#  via Playwright and symlink to a stable path)
# ====================================================

RUN if [ "$TARGETARCH" = "arm64" ]; then \
      mkdir -p /application/bi/dataservice/puppeteer/Chrome && \
      python3 -m pip install --break-system-packages --no-cache-dir playwright && \
      PLAYWRIGHT_BROWSERS_PATH=/opt/ms-playwright \
        python3 -m playwright install --with-deps chromium && \
      ln -sf /opt/ms-playwright/chromium-*/chrome-linux/chrome /application/bi/dataservice/puppeteer/Chrome/chrome && \
      chmod -R a+rX /opt/ms-playwright && \
      apt-get clean && rm -rf /var/lib/apt/lists/*; \
    fi

# ====================================================
# Optional PostgreSQL Installation
# (noble uses the noble-pgdg apt suite instead of bookworm-pgdg)
# ====================================================

RUN if [ "$INSTALL_POSTGRES" = "true" ]; then \
      echo "Installing PostgreSQL 15..." && \
      curl -fsSL https://www.postgresql.org/media/keys/ACCC4CF8.asc | gpg --dearmor -o /usr/share/keyrings/postgresql.gpg && \
      echo "deb [signed-by=/usr/share/keyrings/postgresql.gpg] http://apt.postgresql.org/pub/repos/apt noble-pgdg main" > /etc/apt/sources.list.d/pgdg.list && \
      apt-get update && apt-get install -y --no-install-recommends postgresql-15 postgresql-contrib-15 && \
      service postgresql start && \
      su - postgres -c "psql -c \"ALTER USER postgres WITH PASSWORD 'Admin@123';\"" && \
      service postgresql stop && \
      apt-get clean; \
    else \
      echo "Skipping PostgreSQL installation"; \
    fi

RUN apt-get purge -y --auto-remove linux-libc-dev || true
RUN rm -rf /var/lib/apt/lists/*

# Environment Variables
# ====================================================

# .NET 10 runtime is NOT installed here: the aspnet:10.0-noble base image
# already ships the ASP.NET Core 10 runtime (/usr/share/dotnet, on PATH).

EXPOSE 80
EXPOSE 443

COPY /application .

# ====================================================
# AMD64 Chrome Setup
# ====================================================

RUN if [ "$TARGETARCH" = "amd64" ]; then \
      cd /application/bi/dataservice/puppeteer/Chrome && \
      unzip chrome-linux64.zip && \
      cp -r chrome-linux64/. . && \
      rm -rf chrome-linux64 chrome-linux64.zip && \
      chmod -R 777 /application/bi/dataservice/puppeteer/Chrome; \
    else \
      echo "ARM64 build - using system Chromium"; \
    fi

RUN chmod +x ./entrypoint.sh

CMD ["./entrypoint.sh"]
