# ====================================================
# Chromium for export (PDF / image / PPT) - amd64 and arm64
# (noble's chromium-browser apt package is a snap stub that
#  fails inside containers, so Playwright's Chromium build is
#  used on every architecture: same install path, same
#  version, native binary per TARGETARCH. Installed before
#  COPY so the layer stays cached across app changes.)
# ====================================================

ARG PLAYWRIGHT_VERSION=1.65.0

RUN set -e; \
    case "$TARGETARCH" in amd64|arm64) ;; \
      *) echo "Unsupported TARGETARCH=$TARGETARCH"; exit 1 ;; esac; \
    python3 -m pip install --break-system-packages --no-cache-dir "playwright==${PLAYWRIGHT_VERSION}"; \
    PLAYWRIGHT_BROWSERS_PATH=/opt/ms-playwright \
      python3 -m playwright install --with-deps --no-shell chromium; \
    chmod -R a+rX /opt/ms-playwright; \
    apt-get clean; rm -rf /var/lib/apt/lists/*

# ====================================================

# ====================================================

      curl -fsSL https://www.postgresql.org/media/keys/ACCC4CF8.asc | gpg --dearmor -o /usr/share/keyrings/postgresql.gpg && \
      echo "deb [signed-by=/usr/share/keyrings/postgresql.gpg] http://apt.postgresql.org/pub/repos/apt noble-pgdg main" > /etc/apt/sources.list.d/pgdg.list && \
      apt-get update && apt-get install -y --no-install-recommends postgresql-15 postgresql-contrib-15 && \
      service postgresql start && \
      su - postgres -c "psql -c \"ALTER USER postgres WITH PASSWORD 'Admin@123';\"" && \
      service postgresql stop && \
      apt-get clean; \
    else \
      echo "Skipping PostgreSQL installation"; \

# ====================================================
# Link Playwright Chromium to the path export uses
# (Exporting.PuppeteerPath = .../puppeteer/Chrome/chrome).
# The binary folder differs per arch (chrome-linux on amd64,
# chrome-linux-arm64 on arm64), so resolve it with find, then
# fail the build unless Chromium can actually print a PDF.
# The bundled chrome-linux64.zip is not used in the image.
# ====================================================

RUN set -e; CHROME_DIR=/application/bi/dataservice/puppeteer/Chrome; \
    BIN=$(find /opt/ms-playwright -path '*/chromium-*/chrome-linux*/chrome' -type f | head -n1); \
    [ -n "$BIN" ] && [ -x "$BIN" ] || { echo "ERROR: Playwright Chromium not found"; exit 1; }; \
    mkdir -p "$CHROME_DIR"; rm -rf "$CHROME_DIR"/*; \
    ln -sf "$BIN" "$CHROME_DIR/chrome"; \
    chmod -R a+rX "$CHROME_DIR"; \
    "$CHROME_DIR/chrome" --headless=new --no-sandbox --disable-gpu --disable-dev-shm-usage \
      --print-to-pdf=/tmp/smoke.pdf "data:text/html,<h1>ok</h1>" >/dev/null 2>&1 || true; \
    test -s /tmp/smoke.pdf || { echo "ERROR: Chromium smoke test failed ($TARGETARCH)"; exit 1; }; \
    rm -f /tmp/smoke.pdf; \
    echo "Chromium OK ($TARGETARCH): $("$CHROME_DIR/chrome" --version)"

RUN chmod +x ./entrypoint.sh

CMD ["./entrypoint.sh"]
